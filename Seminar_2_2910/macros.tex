%% ---- Review comments ------------------------------------------------
%% \TODO{...}         something still missing in the paper
%% \cmt{Name}{...}    a reviewer's comment; delete once resolved
%% \new{...}          a short piece of text you added and want others to check
%% \begin{added}...\end{added}
%%                    a whole new paragraph or list (may span paragraphs)
%% \addedtag          put at the start of a caption to flag a new table/figure
%% \fillin{...}       a slot the team must fill (number, name, link)
%% Set \reviewfalse to hide all marks in one go; text inside \new{} and
%% the added environment stays, in black.
\newif\ifreview \reviewtrue
\ifreview
  \newcommand{\TODO}[1]{\textcolor{red}{\textbf{[TODO: #1]}}}
  \newcommand{\cmt}[2]{\textcolor{blue}{\textbf{[#1:} #2\textbf{]}}}
  \newcommand{\new}[1]{\textcolor{teal}{#1}}
  \newenvironment{added}{\color{teal}}{}
  \newcommand{\addedtag}{\textcolor{teal}{\textbf{[new]}}~}
  \newcommand{\fillin}[1]{\colorbox{orange!25}{\textbf{#1}}}
  \newcommand{\hcrev}[1]{\textcolor{magenta!70!black}{#1}}
  \newcommand{\revtag}{\textcolor{magenta!70!black}{\textbf{[revised]}}~}
\else
  \newcommand{\TODO}[1]{}
  \newcommand{\cmt}[2]{}
  \newcommand{\new}[1]{#1}
  \newenvironment{added}{}{}
  \newcommand{\addedtag}{}
  \newcommand{\fillin}[1]{#1}
\fi

%% Mapping-data review: cyan marks disappear in the clean version.
